1use ariel_os_sensors::sensor::{Mode, SetModeError, State};
2use portable_atomic::{AtomicU8, Ordering};
3
4/// A helper to store [`State`] as an atomic.
5#[derive(Default)]
6pub struct AtomicState {
7    state: AtomicU8,
8}
9
10impl AtomicState {
11    /// Creates a new [`AtomicState`].
12    #[must_use]
13    pub const fn new(state: State) -> Self {
14        // Make sure `State` fits into a `u8`.
15        const {
16            assert!(core::mem::size_of::<State>() == core::mem::size_of::<u8>());
17        }
18
19        Self {
20            state: AtomicU8::new(state as u8),
21        }
22    }
23
24    /// Returns the current state.
25    #[expect(clippy::missing_panics_doc, reason = "cannot actually panic")]
26    pub fn get(&self) -> State {
27        // NOTE(no-panic): cast cannot fail because the integer value always comes from *us*
28        // internally casting `State`.
29        State::try_from(self.state.load(Ordering::Acquire)).unwrap()
30    }
31
32    /// Sets the current state.
33    pub fn set(&self, state: State) {
34        self.state.store(state as u8, Ordering::Release);
35    }
36
37    /// Sets the state from a requested mode and returns the previous state.
38    ///
39    /// # Errors
40    ///
41    /// Returns [`SetModeError::Uninitialized`] if the current state is [`State::Uninitialized`].
42    #[expect(clippy::missing_panics_doc, reason = "cannot actually panic")]
43    pub fn set_mode(&self, mode: Mode) -> Result<State, SetModeError> {
44        let new_state = State::from(mode);
45
46        // Set the mode if the current state is not uninitialized
47        self.state
48            .fetch_update(Ordering::Release, Ordering::Acquire, |s| {
49                if s == State::Uninitialized as u8 {
50                    None
51                } else {
52                    Some(new_state as u8)
53                }
54            })
55            .map(|s| {
56                // NOTE(no-panic): cast cannot fail because the integer value always comes from *us*
57                // internally casting `State`.
58                State::try_from(s).unwrap()
59            })
60            .map_err(|_| SetModeError::Uninitialized)
61    }
62}

64#[cfg(test)]
65mod tests {
66    use super::*;
67
68    #[test]
69    fn check_type_sizes() {
70        assert_eq!(size_of::<AtomicState>(), size_of::<u8>());
71        assert_eq!(align_of::<AtomicState>(), 1);
72    }
73
74    #[test]
75    fn preserve_state() {
76        let state = AtomicState::new(State::Uninitialized);
77
78        state.set(State::Enabled);
79        assert_eq!(state.get(), State::Enabled);
80
81        assert!(matches!(state.set_mode(Mode::Sleeping), Ok(State::Enabled)));
82        assert_eq!(state.get(), State::Sleeping);
83
84        assert!(matches!(state.set_mode(Mode::Enabled), Ok(State::Sleeping)));
85        assert_eq!(state.get(), State::Enabled);
86
87        assert!(matches!(state.set_mode(Mode::Sleeping), Ok(State::Enabled)));
88        assert_eq!(state.get(), State::Sleeping);
89    }
90
91    #[test]
92    fn uninitialized_err() {
93        let state = AtomicState::new(State::Uninitialized);
94
95        assert!(matches!(
96            state.set_mode(Mode::Sleeping),
97            Err(SetModeError::Uninitialized)
98        ));
99
100        assert!(matches!(
101            state.set_mode(Mode::Enabled),
102            Err(SetModeError::Uninitialized)
103        ));
104
105        assert!(matches!(
106            state.set_mode(Mode::Disabled),
107            Err(SetModeError::Uninitialized)
108        ));
109    }
