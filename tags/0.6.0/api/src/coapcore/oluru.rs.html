1//! An owned heapless container with capacity N that maintains order both through properties of its
2//! entries and by time of access -- a mix between a priority queue and an LRU cache.
3//!
4//! This was inspired by the [`uluru`](https://github.com/servo/uluru) crate. Adapting that crate
5//! to support priority levels proved to be impractical -- and many properties have changed since.
6//!
7//! See the [`OrderedPool`] documentation for details.
8//!
9//! # Terminology
10//!
11//! The ordering imposed on the cache entries is determined by a priority (see [`PriorityLevel`]).
12//! Throughout the documentation, "high" and "low" indicate priorities, whereas "small" and "large"
13//! indicate numeric values (including of priorities, where "high" corresponds to "small" and "low"
14//! to "large").
15#![expect(clippy::indexing_slicing, reason = "module is scheduled for overhaul")]
16#![expect(
17    clippy::doc_paragraphs_missing_punctuation,
18    reason = "module is scheduled for overhaul"

24/// Required trait on [`OrderedPool`] entries that allows ordering.
25///
26/// Priority levels follow the conventions common with schedulers: 0 is the highest priority, and
27/// will only get evicted if the cache is full with other entries of the same priority. Larger
28/// numeric values indicate increasingly lower priority.
29pub trait PriorityLevel {
30    /// Calculate the priority of the instance
31    ///
32    /// An instance's priority level may change while being mutated; [`OrderedPool`] will account for
33    /// that.
34    ///
35    /// The level should not change due to global effects (or internal mutability, if shared access
36    /// is later implemented). If it does, the ordering of an [`OrderedPool`] containing it may become
37    /// arbitrary, even after the element whose level changed has been removed.
38    fn level(&self) -> usize;
39}

41/// An owned heapless container with capacity `N` that maintains order both through properties of its
42/// entries and by time of access.
43///
44/// Operations that are supposed to be fast are:
45///
46/// * Finding an element by iteration
47/// * Moving that element to the front of its relevant level
48///
49/// There is no remove an item; instead, the `&mut T` of [`Self::lookup`] can be replaced with a
50/// low-priority placeholder value. (In fact, future iterations may require that such a value
51/// exists and is [`Default`]).
52///
53/// # Usage
54///
55/// ```
56/// use coapcore::oluru::{OrderedPool, PriorityLevel};
57///
58/// #[derive(Debug)]
59/// struct MyValue {
60///     id: u32,
61///     name: Option<&'static str>,
62/// }
63///
64/// impl PriorityLevel for MyValue {
65///     fn level(&self) -> usize {
66///         if self.name.is_some() {
67///             0
68///         } else {
69///             1
70///         }
71///     }
72/// }
73///
74/// // A cache with a capacity of three.
75/// type MyCache = OrderedPool<MyValue, 3, 2>;
76///
77/// // Create an empty cache, then insert some items.
78/// let mut cache = MyCache::new();
79/// cache.insert(MyValue { id: 1, name: Some("Mercury") });
80/// cache.insert(MyValue { id: 2, name: Some("Venus") });
81/// cache.insert(MyValue { id: 3, name: None });
82///
83/// let item = cache.lookup(|x| x.id == 1, |x| format!("Found {}", x.name.unwrap_or("unnamed object")));
84/// assert_eq!(item.unwrap().as_str(), "Found Mercury");
85///
86/// // If the cache is full, inserting a new item evicts one item.
87/// //
88/// // While Venus (ID 2) was used least recently, it has higher priority than the no-name planet
89/// // with index 3, so that gets evicted first instead.
90/// let returned = cache.insert(MyValue { id: 4, name: Some("Mars") });
91/// assert!(returned.expect("Pool was full").is_some_and(|item| item.id == 3));
92/// ```
93///
94/// # Implementation
95///
96/// Right now, this is implemented as a separate entry vector and an index vector, where the latter
97/// is often rotated internally. Future changes may change this to only be a single list, using a
98/// doubly linked list, and keeping head indices of each level (which is why the number of levels
99/// `L` is part of the type).
100///
101/// The value list following the style of a `Vec` means that popping elements from anywhere but the
102/// tail is costly, which it should better not be; a slab allocator style would improve that.
103///
104/// ## Terminology
105///
106/// A "position" is a key to `.sorted`. An "index" is a key to `.entries` (and thus a value of
107/// `.sorted`).
108///
109/// ## Invariants
110///
111/// Before and after any public function, these hold:
112///
113/// * `.sorted` has the same length as `.entries`
114/// * `.sorted` is a permutation of `.entries`' (and thus its) index space. Therefore, each of its
115///   values is unique, and is an index into `.entries`.
116/// * If `T::level` is constant, `self.sorted.iter().map(|i| self.entries[i].level())` is sorted.
117#[derive(Debug)]
118pub struct OrderedPool<T: PriorityLevel, const N: usize, const L: usize> {
119    /// Elements without regard for ordering
120    pub entries: ArrayVec<T, N>,
121    /// A sorted list of indices into entries: high priority first, ties broken by recentness
122    pub sorted: ArrayVec<u16, N>,
123}
124
125impl<T: PriorityLevel, const N: usize, const L: usize> OrderedPool<T, N, L> {
126    /// Create an empty cache.
127    #[must_use]
128    pub const fn new() -> Self {
129        assert!(N < u16::MAX as usize, "Capacity overflow");
130        // Clipping levels to u16 because they may be stored if the implementation changes.
131        assert!(L < u16::MAX as usize, "Level overflow");
132        OrderedPool {
133            entries: ArrayVec::new_const(),
134            sorted: ArrayVec::new_const(),
135        }
136    }
137
138    /// Iterate over items from highest priority to lowest, most recently used first.
139    ///
140    /// If the function `f_test` (which receives a shared reference to the entry) returns
141    /// `Some(r)`, `f_use` is called with a *mutable* reference to the item as well as the first
142    /// function's result. That item is regarded as "used" and thus shifted to the front, and the
143    /// second function's return value is passed on.
144    ///
145    /// This differs from `uluru` in two ways:
146    /// * There is no `find()` method: As the level may change through mutation, we can not hand
147    ///   out a `&mut T` unless we can sure to process any level changes when it is returned. (A
148    ///   linear type drop guard wrapper may afford that, but is not available in Rust at the time
149    ///   of writing)
150    /// * The callback is split in a test part and a use part, which ensures that elements that are
151    ///   not looked up do not get mutated; only the selected item is mutated and will then be
152    ///   sorted in correctly.
153    pub fn lookup<Ftest, Fuse, R>(&mut self, mut f_test: Ftest, f_use: Fuse) -> Option<R>
154    where
155        Ftest: FnMut(&T) -> bool,
156        Fuse: FnOnce(&mut T) -> R,
157    {
158        for (position, &index) in self.sorted.iter().enumerate() {
159            if f_test(&self.entries[usize::from(index)]) {
160                let r = f_use(&mut self.entries[usize::from(index)]);
161                self.touch(position);
162                return Some(r);
163            }
164        }
165        None
166    }
167
168    /// Inserts an element.
169    ///
170    /// If the new element's priority is lower than the lowest in the queue, it is returned as an
171    /// Err. Otherwise, the element is inserted, and any dropped lower priority element is
172    /// returned in the Ok value.
173    // Dead code is only allowed and not expected because of interactions testing.
174    #[allow(
175        dead_code,
176        reason = "Need for this function is unclear, but it is covered in tests and docs, and may easily be needed again as coapcore is being refactored."
177    )]
178    pub fn insert(&mut self, new: T) -> Result<Option<T>, T> {
179        let new_index = self.entries.len();
180        if new_index < N {
181            self.entries.push(new);
182            self.sorted.push(
183                new_index
184                    .try_into()
185                    .expect("Range is checked at construction time"),
186            );
187            self.touch(new_index);
188            Ok(None)
189        } else {
190            let last_slot = &mut self.entries
191                [usize::from(*self.sorted.last().expect("Full array is not empty"))];
192            let last_level = last_slot.level();
193            let new_level = new.level();
194            debug_assert!(new_level < L, "Level exceeds limit L={L} in type");
195            if new_level <= last_level {
196                let last = core::mem::replace(last_slot, new);
197                self.touch(N - 1);
198                Ok(Some(last))
199            } else {
200                Err(new)
201            }
202        }
203    }
204
205    // FIXME: It is not fully clear when we would use insert and force_insert -- we'll need some
206    // kind of force_insert to get out of situations where all authenticated connections are really
207    // timing out. But at the same time, if we're being bombarded with bad requests, we should
208    // retain the ability to randomly keep some incoming connection for longer, so that a message 3
209    // that comes through can establish the connection. In the end, we may need some more flexible
210    // policy than just levels and LRU.
211    //
212    // Unless we want to keep track of connections somewhere in parallel, timeouts may also involve
213    // some function called on all present items to mark them as "not used in a long time",
214    // downgrading their priority.
215
216    /// Inserts an element without regard for its level.
217    ///
218    /// The element is inserted unconditionally, and the least priority element is returned by
219    /// value.
220    pub fn force_insert(&mut self, new: T) -> Option<T> {
221        let new_index = self.entries.len();
222        if new_index < N {
223            self.entries.push(new);
224            self.sorted.push(
225                new_index
226                    .try_into()
227                    .expect("Range is checked at construction time"),
228            );
229            self.touch(new_index);
230            None
231        } else {
232            let last_slot = &mut self.entries
233                [usize::from(*self.sorted.last().expect("Full array is not empty"))];
234            let last = core::mem::replace(last_slot, new);
235            self.touch(N - 1);
236            Some(last)
237        }
238    }
239
240    fn touch(&mut self, position: usize) {
241        let level = self.entries[usize::from(self.sorted[position])].level();
242        debug_assert!(level < L, "Level exceeds limit L={L} in type");
243        let mut new_position = position;
244        // Common case: level stayed the same, but we move to front; also applicable when numeric
245        // level decreased
246        while new_position
247            .checked_sub(1)
248            .is_some_and(|n| self.entries[usize::from(self.sorted[n])].level() >= level)
249        {
250            new_position -= 1;
251        }
252        if new_position == position {
253            // Level may instead have increased
254            while new_position < self.sorted.len() - 1
255                && self.entries[usize::from(self.sorted[new_position + 1])].level() < level
256            {
257                new_position += 1;
258            }
259            // Push our entry out left and in right in the rear
260            if new_position != position {
261                self.sorted[position..=new_position].rotate_left(1);
262            }
263        } else {
264            // Push our entry out right and in left in the front
265            self.sorted[new_position..=position].rotate_right(1);
266        }
267    }
268
269    /// Returns an iterator visiting all items in arbitrary order.
270    pub(crate) fn iter(&self) -> impl Iterator<Item = &T> {
271        self.entries.iter()
272    }
273}
