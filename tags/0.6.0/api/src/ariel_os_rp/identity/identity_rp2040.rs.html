5pub struct DeviceId(u64);
6
7// Manufacturer IDs for which the memory chips return (at least) eight unique bytes as unique ID.
8// NOTE: this can be expanded with other manufacturers; currently the type of memory chip is
9// constrained by the second-stage bootloader (boot2).
10const KNOWN_JEDEC_MANUFACTURED_ID: &[u8] = &[
11    0xef, // Used by Winbond chips.
12];
13
14impl ariel_os_embassy_common::identity::DeviceId for DeviceId {
15    type Bytes = [u8; 8];
16
17    fn get() -> Result<Self, impl core::error::Error> {
18        // The flash size does not matter here.
19        const FLASH_SIZE: usize = 0;
20
21        let mut unique_id: Self::Bytes = Default::default();
22
23        // This critical section takes less than 200 µs on a RPi Pico.
24        let jedec_id = critical_section::with(|_| {
25            // SAFETY: the peripheral is stolen and used in a critical section.
26            let flash_peripheral = unsafe { embassy_rp::peripherals::FLASH::steal() };
27            let mut flash =
28                embassy_rp::flash::Flash::<_, _, FLASH_SIZE>::new_blocking(flash_peripheral);
29
30            flash
31                .blocking_jedec_id()
32                .map_err(|_| Error::MemoryChipIdInaccessible)
33        })?;
34        debug!("JEDEC ID: {:#x}", jedec_id);
35
36        // The JEDEC ID is fetched with the 0x9f SPI instruction, which returns three bytes.
37        #[expect(clippy::cast_possible_truncation, reason = "the truncation is desired")]
38        let manufacturer_id = (jedec_id >> 16) as u8;
39
40        if !KNOWN_JEDEC_MANUFACTURED_ID.contains(&manufacturer_id) {
41            return Err(Error::UnsupportedMemoryChip);
42        }
43
44        // This critical section takes less than 200 µs on a RPi Pico.
45        critical_section::with(|_| {
46            // SAFETY: the peripheral is stolen and used in a critical section.
47            let flash_peripheral = unsafe { embassy_rp::peripherals::FLASH::steal() };
48            let mut flash =
49                embassy_rp::flash::Flash::<_, _, FLASH_SIZE>::new_blocking(flash_peripheral);
50
51            flash
52                .blocking_unique_id(&mut unique_id)
53                .map_err(|_| Error::MemoryChipIdInaccessible)
54        })?;
55
56        let unique_id = u64::from_be_bytes(unique_id);
57
58        Ok::<_, Error>(Self(unique_id))
59    }
60
61    fn bytes(&self) -> Self::Bytes {
62        self.0.to_le_bytes()
63    }
64}
65
66#[derive(Debug, Clone, PartialEq, Eq)]
67#[cfg_attr(feature = "defmt", derive(defmt::Format))]
68pub enum Error {
69    MemoryChipIdInaccessible,
70    UnsupportedMemoryChip,
71}
72
73impl core::fmt::Display for Error {
74    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
75        match self {
76            Self::MemoryChipIdInaccessible => {
77                write!(f, "the ID of the memory chip is accessible")
78            }
79            Self::UnsupportedMemoryChip => {
80                write!(f, "the memory chip is unsupported to fetch its unique ID")
81            }
82        }
83    }
84}
