7use ariel_os_embassy_common::gpio::input::InterruptError;
8use embassy_stm32::{OptionalPeripherals, Peri, exti::AnyChannel, gpio::Pin, peripherals};
9use portable_atomic::{AtomicBool, AtomicU16, Ordering};
10
11pub static EXTINT_REGISTRY: ExtIntRegistry = ExtIntRegistry::new();
12
13pub struct ExtIntRegistry {
14    initialized: AtomicBool,
15    used_interrupt_channels: AtomicU16, // 16 channels
16}
17
18impl ExtIntRegistry {
19    // Collect all channel peripherals so that the registry is the only one managing them.
20    const fn new() -> Self {
21        Self {
22            initialized: AtomicBool::new(false),
23            used_interrupt_channels: AtomicU16::new(0),
24        }
25    }
26
27    #[expect(clippy::missing_panics_doc)]
28    pub fn init(&self, peripherals: &mut OptionalPeripherals) {
29        peripherals.EXTI0.take().unwrap();
30        peripherals.EXTI1.take().unwrap();
31        peripherals.EXTI2.take().unwrap();
32        peripherals.EXTI3.take().unwrap();
33        peripherals.EXTI4.take().unwrap();
34        peripherals.EXTI5.take().unwrap();
35        peripherals.EXTI6.take().unwrap();
36        peripherals.EXTI7.take().unwrap();
37        peripherals.EXTI8.take().unwrap();
38        peripherals.EXTI9.take().unwrap();
39        peripherals.EXTI10.take().unwrap();
40        peripherals.EXTI11.take().unwrap();
41        peripherals.EXTI12.take().unwrap();
42        peripherals.EXTI13.take().unwrap();
43        peripherals.EXTI14.take().unwrap();
44        peripherals.EXTI15.take().unwrap();
45
46        self.initialized.store(true, Ordering::Release);
47
48        // Do nothing else, just consume the peripherals: they are ours now!
49    }
50
51    /// # Errors
52    ///
53    /// Returns `Err(InterruptError::IntChannelAlreadyUsed)` if the interrupt channel is already in
54    /// use.
55    ///
56    /// # Panics
57    ///
58    /// Will panic if the interrupt channels have not been captured during initialization.
59    pub fn get_interrupt_channel_for_pin<T: Pin>(
60        &self,
61        pin: &Peri<'_, T>,
62    ) -> Result<Peri<'_, AnyChannel>, InterruptError> {
63        // Make sure that the interrupt channels have been captured during initialization.
64        assert!(self.initialized.load(Ordering::Acquire));
65
66        let pin_number = pin.pin();
67
68        // As interrupt channels are mutually exclusive between ports (ie., if channel i has
69        // been bound for pin i of a port, it cannot be used for pin i of another port), we
70        // only check the pin number.
71        // NOTE(ordering): since setting a bit is an idempotent operation, and since we do not
72        // allow clearing them, the ordering does not matter.
73        let was_used = self
74            .used_interrupt_channels
75            .bit_set(pin_number.into(), Ordering::Relaxed);
76
77        if was_used {
78            return Err(InterruptError::IntChannelAlreadyUsed);
79        }
80
81        // They are the same
82        let ch_number = pin_number;
83
84        // NOTE(embassy): ideally we would be using `T::ExtiChannel::steal()` instead of this
85        // match, but Embassy does not provide this.
86        // SAFETY: this function enforces that the same channel cannot be obtained twice,
87        // making sure multiple instances are not used at the same time as the mandatory
88        // `init()` method has collected all channel peripherals beforehand.
89        let ch = match ch_number {
90            0 => unsafe { peripherals::EXTI0::steal() }.into(),
91            1 => unsafe { peripherals::EXTI1::steal() }.into(),
92            2 => unsafe { peripherals::EXTI2::steal() }.into(),
93            3 => unsafe { peripherals::EXTI3::steal() }.into(),
94            4 => unsafe { peripherals::EXTI4::steal() }.into(),
95            5 => unsafe { peripherals::EXTI5::steal() }.into(),
96            6 => unsafe { peripherals::EXTI6::steal() }.into(),
97            7 => unsafe { peripherals::EXTI7::steal() }.into(),
98            8 => unsafe { peripherals::EXTI8::steal() }.into(),
99            9 => unsafe { peripherals::EXTI9::steal() }.into(),
100            10 => unsafe { peripherals::EXTI10::steal() }.into(),
101            11 => unsafe { peripherals::EXTI11::steal() }.into(),
102            12 => unsafe { peripherals::EXTI12::steal() }.into(),
103            13 => unsafe { peripherals::EXTI13::steal() }.into(),
104            14 => unsafe { peripherals::EXTI14::steal() }.into(),
105            15 => unsafe { peripherals::EXTI15::steal() }.into(),
106            _ => unreachable!(),
107        };
108
109        Ok(ch)
110    }
