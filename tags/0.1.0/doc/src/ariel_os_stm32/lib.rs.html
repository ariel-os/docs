74#[doc(hidden)]
75pub fn init() -> OptionalPeripherals {
76    let mut config = Config::default();
77    board_config(&mut config);
78
79    #[cfg(not(capability = "hw/stm32-dual-core"))]
80    let peripherals = embassy_stm32::init(config);
81
82    #[cfg(capability = "hw/stm32-dual-core")]
83    let peripherals = embassy_stm32::init_primary(config, &SHARED_DATA);
84
85    OptionalPeripherals::from(peripherals)
86}

88// TODO: find better place for this
89fn board_config(config: &mut Config) {
90    #[cfg(context = "st-nucleo-wb55")]
91    {
92        use embassy_stm32::rcc::*;
93
94        config.rcc.hsi48 = Some(Hsi48Config {
95            sync_from_usb: true,
96        }); // needed for USB
97        config.rcc.sys = Sysclk::PLL1_R;
98        config.rcc.hse = Some(Hse {
99            freq: embassy_stm32::time::Hertz(32000000),
100            mode: HseMode::Oscillator,
101            prescaler: HsePrescaler::DIV1,
102        });
103        config.rcc.pll = Some(Pll {
104            source: PllSource::HSE,
105            prediv: PllPreDiv::DIV2,
106            mul: PllMul::MUL10,
107            divp: None,
108            divq: None,
109            divr: Some(PllRDiv::DIV2), // sysclk 80Mhz (32 / 2 * 10 / 2)
110        });
111        config.rcc.mux.clk48sel = mux::Clk48sel::HSI48;
112    }
113
114    #[cfg(context = "stm32h755zitx")]
115    {
116        use embassy_stm32::rcc::*;
117
118        config.rcc.hsi = Some(HSIPrescaler::DIV1);
119        config.rcc.csi = true;
120        config.rcc.hsi48 = Some(Hsi48Config {
121            sync_from_usb: true,
122        }); // needed for USB
123        config.rcc.pll1 = Some(Pll {
124            source: PllSource::HSI,
125            prediv: PllPreDiv::DIV4,
126            mul: PllMul::MUL50,
127            divp: Some(PllDiv::DIV2),
128            // Required for SPI (configured by `spi123sel`)
129            divq: Some(PllDiv::DIV16), // FIXME: adjust this divider
130            divr: None,
131        });
132        config.rcc.sys = Sysclk::PLL1_P; // 400 Mhz
133        config.rcc.ahb_pre = AHBPrescaler::DIV2; // 200 Mhz
134        config.rcc.apb1_pre = APBPrescaler::DIV2; // 100 Mhz
135        config.rcc.apb2_pre = APBPrescaler::DIV2; // 100 Mhz
136        config.rcc.apb3_pre = APBPrescaler::DIV2; // 100 Mhz
137        config.rcc.apb4_pre = APBPrescaler::DIV2; // 100 Mhz
138        config.rcc.voltage_scale = VoltageScale::Scale1;
139        // Set SMPS power config otherwise MCU will not powered after next power-off
140        config.rcc.supply_config = SupplyConfig::DirectSMPS;
141        config.rcc.mux.usbsel = mux::Usbsel::HSI48;
142        // Select the clock signal used for SPI1, SPI2, and SPI3.
143        // FIXME: what to do about SPI4, SPI5, and SPI6?
144        config.rcc.mux.spi123sel = mux::Saisel::PLL1_Q; // Reset value
145    }
146
147    // mark used
148    let _ = config;
149}
