27/// SPI bus configuration.
28#[derive(Clone)]
29#[non_exhaustive]
30pub struct Config {
31    /// The frequency at which the bus should operate.
32    pub frequency: Frequency,
33    /// The SPI mode to use.
34    pub mode: Mode,
35    #[doc(hidden)]
36    pub bit_order: BitOrder,
37}
38
39impl Default for Config {
40    fn default() -> Self {
41        Self {
42            frequency: Frequency::F(Kilohertz::MHz(1)),
43            mode: Mode::Mode0,
44            bit_order: BitOrder::default(),
45        }
46    }
47}

49/// SPI bus frequency.
50#[derive(Debug, Copy, Clone, PartialEq, Eq)]
51#[cfg_attr(feature = "defmt", derive(defmt::Format))]
52#[repr(u32)]
53pub enum Frequency {
54    /// Arbitrary frequency.
55    F(Kilohertz),
56}
57
58impl From<Frequency> for Hertz {
59    fn from(freq: Frequency) -> Self {
60        match freq {
61            Frequency::F(kilohertz) => Hertz::khz(kilohertz.to_kHz()),
62        }
63    }
64}

69macro_rules! define_spi_drivers {
70    ($( $interrupt:ident => $peripheral:ident ),* $(,)?) => {
71        $(
72            /// Peripheral-specific SPI driver.
73            pub struct $peripheral {
74                spim: YieldingAsync<BlockingAsync<InnerSpi<'static, Blocking>>>,
75            }
76
77            impl $peripheral {
78                /// Returns a driver implementing [`embedded_hal_async::spi::SpiBus`] for this SPI
79                /// peripheral.
80                #[expect(clippy::new_ret_no_self)]
81                #[must_use]
82                pub fn new(
83                    sck_pin: impl Peripheral<P: SckPin<peripherals::$peripheral>> + 'static,
84                    miso_pin: impl Peripheral<P: MisoPin<peripherals::$peripheral>> + 'static,
85                    mosi_pin: impl Peripheral<P: MosiPin<peripherals::$peripheral>> + 'static,
86                    config: Config,
87                ) -> Spi {
88                    let mut spi_config = embassy_stm32::spi::Config::default();
89                    spi_config.frequency = config.frequency.into();
90                    spi_config.mode = crate::spi::from_mode(config.mode);
91                    spi_config.bit_order = crate::spi::from_bit_order(config.bit_order);
92                    spi_config.miso_pull = gpio::Pull::None;
93
94                    // Make this struct a compile-time-enforced singleton: having multiple statics
95                    // defined with the same name would result in a compile-time error.
96                    paste::paste! {
97                        #[allow(dead_code)]
98                        static [<PREVENT_MULTIPLE_ $peripheral>]: () = ();
99                    }
100
101                    // FIXME(safety): enforce that the init code indeed has run
102                    // SAFETY: this struct being a singleton prevents us from stealing the
103                    // peripheral multiple times.
104                    let spim_peripheral = unsafe { peripherals::$peripheral::steal() };
105
106                    // The order of MOSI/MISO pins is inverted.
107                    let spim = InnerSpi::new_blocking(
108                        spim_peripheral,
109                        sck_pin,
110                        mosi_pin,
111                        miso_pin,
112                        spi_config,
113                    );
114
115                    Spi::$peripheral(Self { spim: YieldingAsync::new(BlockingAsync::new(spim)) })
116                }
117            }
118        )*
119
120        /// Peripheral-agnostic driver.
121        pub enum Spi {
122            $(
123                #[doc = concat!(stringify!($peripheral), " peripheral.")]
124                $peripheral($peripheral)
125            ),*
126        }
127
128        impl embedded_hal_async::spi::ErrorType for Spi {
129            type Error = embassy_stm32::spi::Error;
130        }
131
132        impl_async_spibus_for_driver_enum!(Spi, $( $peripheral ),*);
133    };
134}
